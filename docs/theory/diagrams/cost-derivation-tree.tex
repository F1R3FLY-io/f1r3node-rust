% ll_required_units = ((1+1)+1)+1 = 4.
%
% Source of truth for the §6.1 derivation in cost-accounting-linear-logic.md.

[{$((a\otimes b)\otimes c)\otimes(1\oplus d)$\\ \textbf{cost = 4}}, fill=caControl
  [{$(a\otimes b)\otimes c$\\ cost 3}, fill=caSupply

  [{$1\oplus d$\\ cost = cost(chosen) = 1}, fill=caSupply, edge label={node[midway,right,font=\sffamily\tiny]{$\oplus$ pays only the chosen branch}}

    [{$d$\\ cost 1 (chosen)}, fill=caPool]]]
\end{forest}
